3. DIAGRAMME DE SÉQUENCE : SIGNALEMENT D'INCIDENT AVEC IA (YOLO)
======================================================================

@startuml
autonumber

actor "Citoyen (App Flutter)" as Citoyen
participant "API Gateway (Spring Security)" as API
participant "Service Incident" as IncService
participant "Service IA (YOLOv8)" as IAService
database "Base de Données (PostGIS)" as DB
actor "Administrateur (Email Notification)" as Admin

Citoyen -> API : Soumettre signalement (Photo + Coordonnées GPS)
activate API
API -> API : Valider le JWT Token Keycloak
API -> IncService : Transmettre la requête de signalement
activate IncService
IncService -> IAService : Envoyer l'image pour prédiction
activate IAService
IAService --> IncService : Retourner la catégorie détectée (ex: Nid-de-poule)
deactivate IAService
IncService -> DB : Enregistrer l'incident (statut='A_MODERER', Point)
activate DB
DB --> IncService : Confirmation d'enregistrement
deactivate DB
IncService -> Admin : Envoyer notification Email (Alerte Nouvel Incident)
IncService --> Citoyen : Confirmer la création du signalement & afficher sur la carte
deactivate IncService
deactivate API
@endum;


======================================================================
4. DIAGRAMME DE SÉQUENCE : RÉSOLUTION & MODÉRATION DE L'INCIDENT
======================================================================

@startuml
autonumber

actor "Agent de Département" as Agent
participant "Système (Backend Spring)" as App
actor Administrateur as Admin
database "Base de Données (PostGIS)" as DB
actor "Citoyen (Alerte/Points)" as Citoyen

Agent -> App : Déclarer résolu + téléverser photo preuve
activate App
App -> DB : Mettre à jour statut = 'EN_COURS_DE_VALIDATION'
App -> DB : Ajouter l'événement à la Timeline de l'incident
App --> Admin : Notifier l'administrateur (Demande de validation)
deactivate App

note over Admin, App : L'Administrateur examine la preuve

alt Preuve Acceptée
    Admin -> App : Valider la résolution (Accepter)
    activate App
    App -> DB : Mettre à jour statut = 'RESOLU'
    App -> DB : Ajouter événement 'Résolution validée' à la Timeline
    App -> DB : Créditer les points d'expérience au Citoyen initiateur
    App --> Citoyen : Notification : "Incident résolu (+50 XP)"
    deactivate App
else Preuve Rejetée
    Admin -> App : Rejeter la résolution (avec motif obligatoire)
    activate App
    App -> DB : Rétablir statut = 'A_TRAITER'
    App -> DB : Ajouter l'événement 'Résolution rejetée' + motif à la Timeline
    App --> Agent : Notifier l'agent du rejet avec le motif de correction
    deactivate App
end
@endum;

5. DIAGRAMME D'ARCHITECTURE SYSTÈME GLOBALE (5 COUCHES)

@startuml
skinparam packageStyle rectangle
skinparam componentStyle uml2
skinparam roundCorner 10
skinparam Shadowing false

skinparam node {
    BorderColor #2C3E50
    BackgroundColor #ECF0F1
}

package "Couche Présentation (Clients)" {
    [Application Mobile (Flutter)] as Flutter
    [Application Web Admin (Angular 18)] as Angular
}

package "Couche Sécurité" {
    database "Keycloak OIDC / SSO" as Keycloak
}

package "Couche Services (Backend)" {
    [API Gateway & Service Métier\n(Spring Boot 3.2)] as SpringBoot
}

package "Couche IA" {
    [Microservice IA\n(Python YOLOv8 / FastAPI)] as YOLO
}

package "Couche Données (Persistance)" {
    database "PostgreSQL\n(Extension PostGIS)" as PostGIS
}

' Relations et flux de données
Flutter -down-> Keycloak : " 1. Authentification (OAuth2 / PKCE)"
Angular -down-> Keycloak : " 1. Connexion / Attribution Rôles"

Flutter -right-> SpringBoot : " 2. Requêtes REST (Bearer Token JWT)"
Angular -left-> SpringBoot : " 2. Requêtes REST (Bearer Token JWT)"

SpringBoot -up-> Keycloak : " 3. Validation de Token public"

SpringBoot -down-> YOLO : " 4. Transmission Image pour Détection (REST API)"
YOLO -up-> SpringBoot : " 5. Retour Catégorie & Métadonnées (JSON)"

SpringBoot -down-> PostGIS : " 6. Opérations SQL / Spatiales (JPA & Hibernate)"

@endum
